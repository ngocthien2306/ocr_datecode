SSH TUNNEL — truy cap he thong tu may Mac
=========================================
Chay lenh nay TREN MAY MAC (khong phai tren workstation).
100.116.14.127 la IP Tailscale cua workstation (msi-MS-7E06).


### Lenh chinh (copy dong nay)

ssh -N -L 5173:localhost:5173 -L 8000:localhost:8000 -L 8001:localhost:8001 -L 8002:localhost:8002 msi@100.116.14.127


### Ban co keepalive (dung khi tunnel bi treo neu de lau)

ssh -N -o ServerAliveInterval=30 -o ServerAliveCountMax=3 \
    -L 5173:localhost:5173 \
    -L 8000:localhost:8000 \
    -L 8001:localhost:8001 \
    -L 8002:localhost:8002 \
    msi@100.116.14.127

### Khoi dong tung service TREN WORKSTATION

# backend
cd ~/Source/ocr_datecode/backend
conda activate vision
python3 -m uvicorn app.main:app --port 8000 --host 0.0.0.0

# frontend
cd ~/Source/ocr_datecode/frontend-ts
npm run dev

# ai_services (camera)
cd ~/Source/ocr_datecode/ai_services
conda activate vision
python3 camera_management_service.py

# anomaly_service
cd ~/Source/ocr_datecode/anomaly_service
conda activate anomaly_service
python -m app.main

# ocr_service — chay tay
cd ~/Source/ocr_datecode/ocr_service
conda activate ocr_train
python -m app.main

# ocr_service — chay bang systemd (khuyen dung)
sudo systemctl start   ocr-training      # hoac restart / stop / status
sudo journalctl -u ocr-training -f       # log
tail -f ~/Source/ocr_datecode/logs/ocr_service.log

# Cai unit lan dau — CHON MOT TRONG HAI:
#
# (a) chi cai rieng ocr-training, KHONG dung cac service dang chay:
sudo cp ~/Source/ocr_datecode/scripts/services/ocr-training.service \
        /etc/systemd/system/ocr-training.service
sudo sed -i "s|%h|$HOME|g" /etc/systemd/system/ocr-training.service
sudo systemctl daemon-reload
sudo systemctl enable --now ocr-training
#
# (b) chay lai installer day du — LUU Y: no xoa va tao lai ocr-backend +
#     ocr-all roi RESTART ca stack (co ca buoc camera check), nen dung khi
#     day chuyen dang nghi:
bash ~/Source/ocr_datecode/scripts/setup_systemd_services.sh

  OCRModelType.CUSTOM.

  backend/.env. Lech thi moi endpoint tra 401 hang loat (khong phai loi
  dang nhap, la loi token khong verify duoc).

- Train OCR khong can tat ai_services: do thuc te 2.6 GB (batch 32) + 3.5 GB
