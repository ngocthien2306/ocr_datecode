# torch/torchvision: install the CUDA build matching this machine's driver
#   FIRST, separately, e.g.:
#   pip install torch==2.5.1 torchvision==0.20.1 --index-url https://download.pytorch.org/whl/cu121
# NOTE (2026-07-28): the pin above does NOT work on Blackwell GPUs (RTX 50-
#   series, sm_120, e.g. RTX 5060 Ti) — torch 2.5.1+cu121 has no compiled
#   kernels past sm_90, so torch.cuda.is_available() reports True but any
#   real op crashes with "CUDA error: no kernel image is available for
#   execution on the device". On Blackwell, install the cu130 build instead
#   (matches anomalib's own cu130 extra, which requires torch>=2.9.0):
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu130
# then install the rest of this file.
anomalib==2.5.0

# ── ONNX Runtime (verify-tensorrt / export-tensorrt endpoints) ─────────────
# onnxruntime-gpu's CUDA/TensorRT execution providers are built against
# CUDA 12 / cuDNN 9, independent of whatever CUDA the torch build above
# uses (CUDA 13 as of the cu130 pin) -- pip install these alongside it,
# NOT instead of it:
#   pip install onnxruntime-gpu nvidia-cublas-cu12 nvidia-cudnn-cu12 \
#       nvidia-cufft-cu12 nvidia-curand-cu12 nvidia-cusolver-cu12 \
#       nvidia-cusparse-cu12 nvidia-nvjitlink-cu12 nvidia-cuda-runtime-cu12
# Their .so files are fully soname-versioned (libcublas.so.12 vs the CUDA 13
# build's .so.13) so both coexist on LD_LIBRARY_PATH without clobbering.
onnxruntime-gpu==1.23.2
